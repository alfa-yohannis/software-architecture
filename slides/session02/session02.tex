\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{10pt}
  \begin{enumerate}
    \item Menjelaskan pembagian tanggung jawab keempat \textit{layer} beserta arah
          ketergantungannya.
    \item Membedakan \textit{closed layer} dari \textit{open layer}, lalu menentukan
          kapan masing-masing layak dipilih.
    \item Mengukur kedalaman \textit{call stack}, \textit{layer violation}, dan
          selisih waktu antara kedua bentuk.
  \end{enumerate}
\end{frame}

\begin{frame}{{\Large Pendahuluan}}
  \vspace{8pt}
  \begin{itemize}
    \item \textit{Layered architecture} menata aplikasi menjadi beberapa
          \textit{layer} yang bertumpuk.
    \item Setiap \textit{layer} mengerjakan satu jenis urusan teknis, lalu memanggil
          \textit{layer} di bawahnya.
    \item Bentuk ini menjadi pilihan pertama bagi sebagian besar sistem, dan
          sering dipakai tanpa pertimbangan panjang.
  \end{itemize}

  \vspace{8pt}
  {\small Alasan pemakaiannya sering bersifat organisasi, bukan teknis.
  Pembagian \textit{layer} mengikuti pembagian kerja yang sudah ada di banyak tim,
  yaitu ada yang mengerjakan antarmuka, ada yang mengerjakan aturan bisnis, dan
  ada yang mengerjakan basis data.}

  \vspace{6pt}
  {\small Kemudahan itu ada harganya. Ketika aplikasi membesar, satu fitur baru
  sering menyentuh keempat \textit{layer} sekaligus.}
\end{frame}

\begin{frame}{{\Large Dua Konsekuensi Aturan Tersebut}}
  \vspace{10pt}
  \begin{itemize}
    \item Satu \textit{layer} dapat diganti tanpa menyentuh \textit{layer} lain, selama
          bentuk panggilannya tetap. Jadi, tidak perlu mengubah \textit{layer} lain.
    \item Permintaan sederhana tetap harus melewati seluruh \textit{layer}, walaupun
          sebagian \textit{layer} tidak mengerjakan apa pun.
  \end{itemize}

  \vspace{10pt}
  {\small Konsekuensi kedua menimbulkan pertanyaan berikutnya, yaitu apakah
  setiap permintaan benar-benar wajib melewati semua \textit{layer}.}
\end{frame}

\begin{frame}{{\Large Architecture Sinkhole}}
  \vspace{10pt}
  \begin{itemize}
    \item Permintaan mengalir melewati beberapa \textit{layer} tanpa satu pun \textit{layer}
          mengerjakan sesuatu.
    \item Keadaan ini hampir tidak mungkin dihapus, sehingga yang diusahakan
          adalah menekan jumlahnya.
    \item Yang diperiksa adalah apakah \textit{business layer} lebih sering meneruskan
          panggilan daripada mengerjakan sesuatu.
    \item \textit{Layer} mana yang tertutup dan mana yang terbuka wajib dicatat dan
          disepakati satu tim.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{{\Large Pass-through yang Wajar dan yang Tidak}}
  \vspace{5pt}
\begin{lstlisting}[style=PythonStyle]
def susun_kurs_dasar(self):
  # Kurs dasar diterbitkan untuk umum, meniru JISDOR.
  # Tidak ada tuntutan bisnis, sehingga murni pass-through.
  return self.repositori.daftar_kurs_dasar()

def susun_kurs_jual(self, token):
  # Kurs jual memuat margin keuntungan bank, dan besar
  # keuntungan itu komersial, sehingga wajib diperiksa.
  pengguna = self.penjaga.periksa_token(token)
  if pengguna is None:
    raise PermissionError("Token tidak dikenali")
  if not self.penjaga.boleh_membaca(pengguna):
    raise PermissionError("Peran tidak berhak membaca")
  return self.repositori.daftar_kurs_jual()
\end{lstlisting}

  \vspace{3pt}
  {\small Keduanya metode pada \texttt{LayananKurs}. Jalan pintas pada yang
  pertama wajar. Pada yang kedua, dua pemeriksaan ikut hilang.}
\end{frame}

\begin{frame}{{\Large Analogi: Tiga Meja dan Satu Cap}}
  \vspace{10pt}
  \begin{itemize}
    \item Surat pengantar harus melewati tiga meja sebelum disetujui.
    \item Meja pertama membaca isinya lalu memutuskan.
    \item Meja kedua dan ketiga hanya membubuhkan cap, lalu meneruskan.
    \item Ketiganya menambah waktu, tetapi hanya satu menambah nilai.
  \end{itemize}

  \vspace{10pt}
  {\small Prosedurnya terlihat rapi di atas kertas, tetapi dua meja terakhir
  adalah \textit{sinkhole}. Itulah yang terjadi pada \textit{layer} yang hanya
  meneruskan panggilan.}
\end{frame}

\begin{frame}{{\Large Kasus: Jalan Pintas yang Dipakai Ulang}}
  \vspace{8pt}
  \begin{enumerate}
    \item Tim membuat halaman perbandingan kurs yang terbuka untuk pengunjung.
    \item Agar cepat selesai, halaman itu langsung membaca \textit{persistence layer}.
    \item Waktu itu pilihannya masih aman, sebab yang tampil hanya kurs
          dasar.
    \item Beberapa bulan berikutnya, kurs jual diminta ikut tampil di halaman
          yang sama.
    \item Nama metodenya tinggal ditukar, sebab jalannya sudah terbuka sejak
          awal.
  \end{enumerate}

  \vspace{8pt}
  {\small Sejak saat itu margin keuntungan bank ikut terbaca siapa saja. Tidak
  ada pesan galat dan tidak ada yang gagal, sehingga tidak ada yang curiga.}
\end{frame}

\begin{frame}{{\Large Ringkasan}}
  \vspace{10pt}
  \begin{itemize}
    \item Empat \textit{layer}, satu aturan, yaitu panggilan hanya mengarah ke bawah.
    \item Aturan itu menahan perubahan di satu \textit{layer}, tetapi memaksa
          permintaan sederhana melewati \textit{layer} yang tidak bekerja.
    \item Metode \textit{pass-through} ditekan jumlahnya, bukan dihapus
          sepenuhnya.
    \item \textit{Open layer} memangkas kode \textit{pass-through}, tetapi menambah
          dua \textit{layer violation}.
    \item Jalan pintas hanya lebih cepat pada 48 dari 100 putaran, dan
          selisihnya 64 kali lebih kecil daripada \textit{range} pengukurannya.
    \item Yang menentukan bukan kecepatan, melainkan ada tidaknya tuntutan
          bisnis yang ikut dilewati.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 1: Menelusuri Jalan Permintaan}}
  \vspace{2pt}
\begin{lstlisting}[language=bash]
# closed, aplikasi_tertutup.py
antarmuka.tampilkan_kurs_jual(token) ----> koneksi.execute(KUERI_KURS_JUAL)
antarmuka.tampilkan_kurs_dasar()     ----> koneksi.execute(KUERI_KURS_DASAR)

# open, aplikasi_terbuka.py
antarmuka.tampilkan_kurs_jual()      ----> koneksi.execute(KUERI_KURS_JUAL)
antarmuka.tampilkan_kurs_dasar()     ----> koneksi.execute(KUERI_KURS_DASAR)
\end{lstlisting}

  \vspace{2pt}
  \begin{enumerate}
    \item Lengkapi isi setiap panah dengan membaca kode, tanpa menjalankannya.
    \item Hitung metode di antara kedua ujung, pemeriksaannya, dan metode
          \textit{pass-through}-nya.
    \item Bandingkan versi \textit{closed} dengan versi \textit{open} pada
          permintaan yang sama.
  \end{enumerate}

  \vspace{2pt}
  {\small Pertanyaan: kedua versi berselisih tepat satu metode. Metode mana
  yang hilang, dan apa yang dibawanya?}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 2: Menemukan \textit{Layer Violation}}}
  \vspace{3pt}
  \vspace{2pt}
  \begin{enumerate}
    \item Salin konstanta \texttt{LAPISAN} pada kedua berkas, lalu baca
          konstruktor setiap kelas beserta anotasi tipenya.
    \item Gambarkan panah dari kelas pemegang ke kelas yang dipegang, lalu
          tandai panah yang melompati satu \textit{layer} atau lebih.
    \item Jalankan kedua aplikasi, lalu bandingkan perlakuannya terhadap token.
  \end{enumerate}

  \vspace{6pt}
  {\small Pertanyaan: bagian kode mana yang tidak melewati \textit{business
  layer} melainkan langsung menuju sumber datanya? Sebutkan kelas, metode, dan
  baris rujukannya.}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 3: Mengukur Biaya Pemeriksaan}}
  \vspace{3pt}
\begin{lstlisting}[language=bash]

\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}
    \item Perhatikan bahwa skrip menjalankan seratus putaran, bukan satu.
    \item Perhatikan baris kemajuan yang dicetak setiap sepuluh putaran, lalu

    \item Baca baris \texttt{\textit{Range} dibagi selisih}, yaitu berapa kali \textit{range} di

          dengan lima puluh, yaitu nilai bila kedua jalur sama cepat.
    \item Jalankan sekali lagi, lalu bandingkan kedua hasilnya.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: hitung penghematannya dalam persen, lalu bandingkan
  dengan tuntutan bisnis yang dilepas.}
