\begin{frame}{{\Large Masuk dan Keluar Menunjuk Arah Panggilan}}
  \vspace{8pt}
  \begin{itemize}\small
    \item Kata masuk dan keluar menunjuk \textbf{arah panggilan}, bukan arah
      data. Data pada keduanya tetap mengalir dua arah.
    \item Port masukan berarti panggilannya datang dari luar. Port keluaran
      berarti panggilannya berangkat dari logika inti.
    \item Port keluaran karena itu tidak terbatas pada pembacaan. Port
      penyimpanan lazimnya memuat \texttt{simpan} dan \texttt{cari} sekaligus.
    \item Setiap sumber menuntut adapternya sendiri, sedangkan satu adapter
      hanya mengisi satu port.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{{\Large Port Ditulis oleh Logika Inti}}
  \vspace{5pt}
\begin{lstlisting}[style=PythonStyle]
class SumberKurs(Protocol):
  # Port keluaran yang dibutuhkan logika inti. Siapa pun yang menyediakan
  # metode ini dapat dipasang, baik memori, basis data, maupun tiruan.

  def nilai(self, kode_asal, kode_tujuan):
    # Mengembalikan nilai kurs, atau None bila pasangannya tidak ada.
\end{lstlisting}

  \vspace{5pt}
  {\small Port hanya menyebut kelakuan yang dibutuhkan, bukan teknologinya.}
\end{frame}

\begin{frame}[fragile]{{\Large Adapter Diterima sebagai Argumen}}
  \vspace{5pt}
\begin{lstlisting}[style=PythonStyle]
def hitung_konversi(sumber, kode_asal, kode_tujuan, nominal):
  # Sumber kurs diterima sebagai argumen, bukan dibuat di dalam fungsi.
  # Karena itu fungsi ini tidak pernah tahu teknologi penyimpanannya.
  if nominal <= 0:
    raise ValueError("Nominal harus lebih besar dari nol")
  kurs = sumber.nilai(kode_asal, kode_tujuan)
  if kurs is None:
    raise LookupError("Kurs tidak tersedia")
  return nominal * kurs
\end{lstlisting}

  \vspace{5pt}
  {\small Bila logika inti membuat adapternya sendiri, seluruh manfaatnya hilang.}
\end{frame}

\begin{frame}{{\Large Analogi: Stopkontak di Dinding}}
  \vspace{10pt}
  \begin{itemize}
    \item Stopkontak menentukan bentuk colokan yang diterimanya.
    \item Stopkontak tidak menentukan alat apa yang dicolokkan.
    \item Kipas angin, lampu, dan pengisi daya sama-sama dapat dipasang.
    \item Dinding tidak perlu diubah setiap kali alatnya berganti.
  \end{itemize}

  \vspace{10pt}
  {\small Port adalah stopkontaknya, adapter adalah colokannya. Logika inti
  tidak boleh membuat adapternya sendiri, sebab dinding yang memilih alat
  bukan lagi stopkontak.}
\end{frame}

\begin{frame}{{\Large Kasus: Klaim yang Gugur karena Satu Baris}}
  \vspace{8pt}
  \begin{enumerate}
    \item Tim menyusun domain, port, dan dua adapter dengan rapi.
    \item Muncul kebutuhan mencatat waktu pembacaan ke berkas log.
    \item Seseorang menambahkan satu baris impor pustaka basis data di
          \texttt{domain.py} agar cepat selesai.
    \item Susunan berkasnya masih terlihat rapi, dan tidak ada galat muncul.
    \item Pemeriksaan otomatis melaporkan satu adapter disebut, dan klaimnya
          gugur.
  \end{enumerate}

  \vspace{8pt}
  {\small Karena itulah klaim yang menyebut angka lebih berguna daripada klaim
  yang hanya menyebut sifat.}
\end{frame}

\end{frame}

\begin{frame}{{\Large Ringkasan}}
  \vspace{10pt}
  \begin{itemize}
    \item Logika inti menyatakan kebutuhannya sebagai port, lalu adapter
          mengisinya. Seluruh panah menunjuk ke dalam.
    \item Satu port dapat diisi banyak adapter, dan keduanya menghasilkan
          jawaban yang sama tanpa mengubah logika inti.
    \item Adapter diserahkan dari luar sebagai argumen, dan pemilihannya
          dipusatkan pada satu berkas perangkai.
    \item Klaim utamanya menyebut angka pasti, yaitu nol teknologi penyimpanan
          yang dikenal logika inti.
    \item Harganya adalah jumlah berkas yang bertambah, sehingga belum tentu
          sepadan untuk aplikasi kecil.
  \end{itemize}
\end{frame}
