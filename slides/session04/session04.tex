\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{10pt}
  \begin{enumerate}
    \item Menjelaskan peran \textit{port} dan \textit{adapter} serta arah
      \textit{dependency} yang membuat \textit{core logic} bebas dari teknologi
      \textit{storage}.
    \item Membedakan \textit{inbound port} dari \textit{outbound port}.
    \item Mengukur \textit{core independence}, \textit{adapter swap cost}, dan
      \textit{testing in isolation}.
  \end{enumerate}
\end{frame}

\begin{frame}{{\Large Pendahuluan}}
  \vspace{8pt}
  \begin{itemize}
    \item Bab 2 menjaga \textit{downward call flow}, dan aturan itu memberi
      \textit{change containment}.
    \item Namun \textit{business layer} tetap memanggil \textit{persistence
      layer}.
    \item Artinya \textit{core logic} tetap punya \textit{storage coupling}.
  \end{itemize}

  \vspace{8pt}
  {\small \textit{Ports and adapters} membalik arah \textit{dependency}.
  \textit{Core logic} tidak lagi memanggil \textit{storage}, melainkan
  menyatakan kebutuhannya sebagai \textit{port}, lalu pihak lain yang
  menyediakannya.}

  \vspace{6pt}
  {\small \textit{Dependency inversion} itu menghasilkan klaim yang dapat
  dibuktikan salah, atau \textit{falsifiable claim}. Isinya angka: teknologi
  \textit{storage} yang disebut \textit{core logic} harus nol.}
\end{frame}

\begin{frame}{{\Large \textit{Inbound} dan \textit{Outbound} Menunjuk
  \textit{Call Direction}}}
  \vspace{8pt}
  \begin{itemize}\small
    \item Kata \textit{inbound} dan \textit{outbound} menunjuk
      \textbf{\textit{call direction}}, bukan arah data. Data pada keduanya
      tetap mengalir dua arah.
    \item \textit{Inbound port} berarti panggilannya datang dari luar.
      \textit{Outbound port} berarti panggilannya berangkat dari \textit{core
      logic}.
    \item Satu \textit{outbound port} biasanya memuat lebih dari satu metode,
      misalnya \texttt{simpan} untuk menulis dan \texttt{cari} untuk membaca.
      Keduanya tetap \textit{outbound}, sebab \textit{core logic} yang memulai.
    \item Setiap \textit{data source} menuntut \textit{adapter}-nya sendiri.
      Satu \textit{adapter} lazimnya memenuhi satu \textit{port}, tetapi boleh
      memenuhi beberapa bila teknologinya sama.
  \end{itemize}
\end{frame}

class RateSource(Protocol):

  def nilai(self, kode_asal, kode_tujuan):
    # Mengembalikan nilai kurs, atau None bila pasangannya tidak ada.
\end{lstlisting}

  \vspace{5pt}
  {\small \textit{Port} hanya menyebut \textit{behavior} yang dibutuhkan, bukan
  teknologinya.}
\end{frame}

\begin{frame}[fragile]{{\Large \textit{Adapter} Diterima sebagai Argumen}}
  \vspace{5pt}
\begin{lstlisting}[style=PythonStyle, basicstyle=\ttfamily\footnotesize]
def hitung_konversi(sumber_kurs, kode_asal, kode_tujuan, nominal):
  # Sumber kurs diterima sebagai argumen, bukan dibuat di dalam fungsi.
  # Karena itu fungsi ini tidak pernah tahu teknologi storage-nya.
  if nominal <= 0:
    raise ValueError("Nominal harus lebih besar dari nol")
  kurs = sumber_kurs.nilai(kode_asal, kode_tujuan)
  if kurs is None:
    raise LookupError("Kurs tidak tersedia")
  return nominal * kurs
\end{lstlisting}

  \vspace{5pt}
  {\small Bila \textit{core logic} membuat \textit{adapter}-nya sendiri,
  seluruh manfaatnya hilang.}
\end{frame}

  \end{itemize}

\begin{frame}{{\Large Analogi: Stopkontak di Dinding}}
  \vspace{10pt}
  \begin{itemize}
    \item Stopkontak menetapkan bentuk lubangnya, misalnya dua lubang pipih.
    \item Alat berkaki tiga tetap dapat dipakai lewat adapter perjalanan yang
      mengubah kaki tiga menjadi dua.
    \item Adapter itulah yang mengetahui bentuk kaki alatnya, sedangkan
      stopkontak tidak.
    \item Ketika alatnya berganti, yang diganti cukup adapternya.
  \end{itemize}

  \vspace{10pt}
  {\small Bentuk lubang itulah \textit{port}-nya, dan pengubah kaki itulah
  \textit{adapter}-nya. \textit{Core logic} tidak boleh membuat
  \textit{adapter}-nya sendiri, sebab stopkontak yang memilih alat berarti
  dindingnya ikut mengenal alat itu.}
\end{frame}

\begin{frame}{{\Large Kasus: Klaim \textit{Falsified} oleh Satu Baris}}
  \vspace{8pt}
  \begin{enumerate}
    \item Tim menyusun domain, \textit{port}, dan dua \textit{adapter} menurut
      aturan pola ini. Klaimnya tercatat sebagai angka, yaitu nol teknologi
      \textit{storage} di dalam \texttt{domain.py}.
    \item Muncul kebutuhan mencatat waktu pembacaan ke \textit{file} log.
    \item Seseorang menambahkan satu baris impor pustaka basis data di
      \texttt{domain.py} agar cepat selesai.
    \item Susunan direktorinya tidak berubah, \textit{unit test} tetap lulus,
      dan tidak ada \textit{error} muncul.
    \item \textit{Automated check} melaporkan satu \textit{adapter} disebut,
      sehingga angkanya bukan nol lagi. Klaim \textit{core independence} pun
      \textit{falsified}.
  \end{enumerate}

\end{frame}

  {\small Klaim seperti itu dapat dibuktikan salah. Itu yang membuatnya

\end{frame}

\begin{frame}{{\Large Ringkasan}}
  \vspace{10pt}
  \begin{itemize}
    \item \textit{Core logic} menyatakan kebutuhannya sebagai \textit{port},
      lalu \textit{adapter} memenuhinya. Setiap panah berangkat dari
      \textit{adapter} menuju \textit{core logic}, tidak pernah sebaliknya.
    \item Satu \textit{port} dapat dipenuhi banyak \textit{adapter}. Setiap
      \textit{adapter} mengembalikan nilai bertipe sama seperti yang diminta
      \textit{port}, sehingga \textit{core logic} tidak berubah.
    \item \textit{Adapter} diserahkan dari luar sebagai argumen, dan
      pemilihannya dipusatkan pada satu \textit{composition root}.
    \item Klaim utamanya berisi angka, yaitu nol teknologi \textit{storage}
      yang boleh dikenal \textit{core logic}.
    \item \textit{Trade-off}-nya adalah jumlah \textit{file} yang bertambah,
      sehingga belum tentu sepadan untuk aplikasi kecil.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 1: \textit{Core Independence}}}
  \vspace{3pt}
\begin{lstlisting}[language=bash]
python periksa_port.py
\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}
    \item Catat baris \texttt{Adapter disebut}.
    \item Buka \texttt{domain.py}, lalu pastikan tidak ada nama teknologi
      \textit{storage} di dalamnya.
    \item Tambahkan \texttt{import sqlite3} pada \texttt{domain.py}, lalu
      jalankan ulang pemeriksaannya.
    \item Batalkan penambahan tersebut, lalu cek ulang lagi, pastikan
      pemeriksaan kembali bersih.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: \texttt{domain.py} tetap mengimpor \texttt{typing}.
  Mengapa impor itu tidak melanggar klaim \textit{core independence}, yaitu nol
  teknologi \textit{storage} yang disebut \texttt{domain.py}?}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 2: \textit{Adapter Swap Cost}}}
  \vspace{3pt}
\begin{lstlisting}[language=bash]
python aplikasi.py memori
python aplikasi.py sqlite
\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}
    \item Pastikan hasil kedua \textit{adapter} sama.
    \item Catat \textit{file} mana saja yang harus diubah untuk berpindah
      \textit{adapter}.
    \item Tulis satu \textit{adapter} ketiga yang membaca kurs dari
      \textit{file} teks biasa.
    \item Catat jumlah \textit{file} dan baris yang berubah agar
      \textit{adapter} ketiga dapat dipakai.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: apakah \texttt{domain.py} ikut berubah ketika
  \textit{adapter} ketiga ditambahkan?}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 3: \textit{Testing in Isolation}}}
  \vspace{3pt}
\begin{lstlisting}[language=bash]
python uji_tanpa_basis_data.py
\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}
    \item Catat hasil ketiga \textit{unit test}-nya.
    \item Hitung jumlah baris kelas \texttt{StubRateSource}.
    \item Tambahkan satu \textit{unit test} baru untuk nominal yang melewati

    \item Hitung berapa kali \textit{unit test} itu membuka koneksi basis data
      lewat \texttt{sqlite3.connect}, membaca \textit{file} dari disk lewat
      \texttt{open}, atau membuka soket lewat \texttt{socket.connect}.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: berapa baris kode yang dibutuhkan untuk menggantikan
  seluruh basis data pada \textit{unit test}?}
\end{frame}
