\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{10pt}
  \begin{enumerate}
    \item Menjelaskan pembagian antara \textit{core} dan \textit{plugin} beserta \textit{contract}
          yang menghubungkan keduanya.
    \item Membedakan \textit{eager activation} dari \textit{lazy activation}, lalu menentukan kapan
          masing-masing layak dipilih.
    \item Mengukur \textit{core independence}, \textit{error isolation}, dan
          \textit{activation cost}.
  \end{enumerate}

  \vspace{10pt}
  {\small Bab ini menyinggung sistem operasi sekilas, lalu menitikberatkan
  arsitektur \textit{plugin} pada perkakas sehari-hari.}
\end{frame}

\begin{frame}{{\Large Pendahuluan}}
  \vspace{8pt}
  \begin{itemize}\small
    \item Microkernel memisahkan sistem menjadi \textit{core} kecil beserta bagian
          tambahan yang dapat dipasang dan dicabut.
    \item Istilahnya lahir di sistem operasi, tetapi bentuk yang paling sering
          ditemui adalah turunannya, yaitu arsitektur \textit{plugin}.
    \item Eclipse, Visual Studio Code, Chrome, dan Firefox memakainya. \textit{Core}
          keempatnya kecil, dan hampir seluruh kemampuan datang dari luar.
  \end{itemize}

  \vspace{8pt}
  {\small Pertanyaannya bukan bagaimana menulis kernel, melainkan bagaimana
  sebuah \textit{core} menerima kemampuan baru tanpa mengenal satu pun penyumbangnya.}
\end{frame}

\begin{frame}[fragile]{{\Large \textit{Contract} yang Ditulis \textit{Core}}}
  \vspace{6pt}
  {\small \textit{Core} menulis \textit{contract}-nya, \textit{plugin} yang memenuhinya. Eclipse
  menyebut \textit{contract} itu \textit{extension point}.}
  \vspace{4pt}
\begin{lstlisting}[style=PythonStyle]
KUNCI_MANIFES = ("nama", "perintah", "keterangan", "modul", "kelas")

class Plugin(Protocol):
  def __init__(self, inti: object) -> None: ...
  def jalankan(self, argumen: list[str]) -> str: ...
\end{lstlisting}
  \vspace{4pt}
  \begin{itemize}\small
    \item \textit{Contract}-nya sengaja kecil, sebab yang gemuk menyulitkan penulis
          \textit{plugin}.
    \item \textit{Manifest} disimpan sebagai \textit{file} JSON terpisah, sehingga dapat dibaca
          tanpa mengimpor kodenya.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{{\Large Contoh Isi \textit{Manifest}}}
  \vspace{5pt}
\begin{lstlisting}[language=json, basicstyle=\ttfamily\footnotesize]

  "keterangan": "Menghitung konversi satu nominal antar mata uang",

  "kelas": "ConversionPlugin"

  \vspace{5pt}
  {\small Kunci \texttt{perintah} dan \texttt{keterangan} dibaca saat mulai
  untuk menyusun daftar, sedangkan \texttt{modul} dan \texttt{kelas} baru
  dipakai ketika perintahnya benar-benar dipanggil.}
\end{frame}

\begin{frame}[fragile]{{\Large \textit{Eager Activation}}}
  \vspace{4pt}
  \begin{itemize}\small
    \item \textit{Eager activation} mengimpor seluruh modul \textit{plugin} saat
          aplikasi mulai, sebelum satu perintah pun dipanggil.
    \item \textit{Lazy activation} menunggu sampai perintahnya dipanggil.
  \end{itemize}

  \vspace{4pt}
\begin{lstlisting}[style=PythonStyle]
def aktifkan_semua(self) -> int:
  jumlah = 0
  for perintah in self.manifes:
    try:
      self.aktifkan(perintah)   # impor, getattr, lalu buat objeknya
      jumlah += 1
    except PluginError:
      pass
  return jumlah
\end{lstlisting}

  \vspace{4pt}
  {\small Keempat modul diimpor walau hanya satu perintah yang dipakai.
  Harganya terukur pada Latihan 3, yaitu \textit{startup time} 3,4 kali lebih
  lambat.}
\end{frame}

\begin{frame}{{\Large \textit{Lazy Activation}: Urutan Panggilan}}
  \vspace{2pt}
  \begin{columns}[c]
    \begin{column}{0.6\textwidth}
      \vspace{10pt}
      \gambarkolom[0.8\textheight]{\includegraphics{sekuens-aktivasi.pdf}}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{itemize}\small
        \item \textit{Manifest} dibaca saat mulai, tanpa satu pun impor.
        \item Modul pluginnya baru diimpor pada pemanggilan pertama.
        \item Objeknya disimpan, sehingga pemanggilan kedua langsung
          menjalankan tanpa impor ulang.
        \item \textit{Plugin} yang tidak pernah dipanggil tidak pernah
          diimpor.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{{\Large Aktivasi Dilindungi \textit{Error Handler}}}
  \vspace{6pt}
\begin{lstlisting}[style=PythonStyle]
def aktifkan(self, perintah: str) -> kontrak.Plugin:
  if perintah in self.plugin_aktif:
    return self.plugin_aktif[perintah]
  manifes = self.manifes[perintah]
  try:
    modul = importlib.import_module(manifes["modul"])
    kelas = getattr(modul, manifes["kelas"])
    plugin = kelas(self)          # core diserahkan ke plugin
  except Exception as kesalahan:
    raise PluginError(f"{manifes['nama']} gagal dimuat: {kesalahan}")
  self.plugin_aktif[perintah] = plugin
  return plugin
\end{lstlisting}
  \vspace{4pt}
  {\small Hasil impor disimpan, sehingga pemanggilan kedua tidak membayar biaya
  yang sama. \textit{Error}-nya ditangkap \textit{error handler} agar satu
  \textit{plugin} rusak tidak menjatuhkan \textit{core}.}
\end{frame}

\begin{frame}{{\Large Analogi: Kantin yang Menyewakan Kios}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Pengelola menyediakan tempat, listrik, air, dan aturan jam buka.
    \item Siapa saja boleh membuka kios selama aturan itu dipenuhi.
    \item Pengelola tidak perlu tahu menu apa yang dijual, sebab yang diurusnya
          hanya daftar penyewa beserta papan namanya.
    \item Satu kios yang tutup tidak menutup kantinnya, sebab pembeli cukup
          berpindah ke kios sebelah.
  \end{itemize}

  \vspace{10pt}
  {\small Aturan jam buka itulah \textit{contract}-nya, papan nama itulah \textit{manifest}-nya, dan
  daftar penyewa itulah \textit{registry}-nya.}
\end{frame}

\begin{frame}{{\Large Kasus: Satu \textit{File} Hilang}}
  \vspace{8pt}
  \begin{enumerate}\small
    \item Sebuah tim memasang lima \textit{plugin}, dan kelimanya diimpor begitu
          aplikasi mulai, yaitu \textit{eager activation}.
    \item Satu \textit{plugin} membaca \textit{file} pengaturan di baris teratas
          modulnya, sehingga \textit{file} itu dibaca sewaktu modulnya diimpor.
    \item Suatu pagi \textit{file} itu terhapus, sehingga impornya gagal.
    \item Impor tersebut berjalan sebelum layar muncul, dan tidak ada
          \textit{error handler} di depannya, sehingga aplikasi berhenti sebelum
          menampilkan apa pun.
  \end{enumerate}

  \vspace{8pt}
  {\small Keempat \textit{plugin} lain sebenarnya tidak rusak. Pertanyaannya,
  mengapa keempatnya ikut berhenti.}
\end{frame}

\begin{frame}{{\Large Ringkasan}}
  \vspace{10pt}
  \begin{itemize}\small
    \item \textit{Core} menulis \textit{contract} dan membaca \textit{manifest},
          sedangkan \textit{plugin} yang memenuhi \textit{contract} itu ditulis
          pihak lain.
    \item Karena \textit{core} tidak menyebut satu nama penyumbang pun, menambah
          kemampuan berarti menambah \textit{file}.
    \item \textit{Manifest} berupa \textit{file} terpisah, sehingga daftar perintah
          terbaca tanpa mengimpor satu modul pun, dan impornya dapat ditunda
          sampai perintahnya dipanggil.
    \item Satu \textit{plugin} yang gagal saat impor dapat menjatuhkan seluruh
          aplikasi bila pemanggilannya tidak dilindungi \textit{error handler}.
  \end{itemize}

  \vspace{10pt}
  {\small Pesan utamanya bukan \textit{core} harus kecil, melainkan \textit{core} harus buta
  terhadap penyumbangnya.}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 1: \textit{Core Independence}}}
  \vspace{3pt}
\begin{lstlisting}[language=bash]
python periksa_inti.py
# Plugin disebut core    : 0 []
\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}\small
    \item Catat baris \texttt{Plugin disebut core} beserta ketiga angka di
          bawahnya.
    \item Salin \texttt{plugins/plugin\_riwayat.py} beserta \textit{manifest}-nya menjadi
          \texttt{plugin\_pajak.py}, lalu ubah nama, perintah, dan kelasnya.
    \item Jalankan ulang keduanya, lalu catat berapa baris \textit{core} yang berubah.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: baris \textit{core} tidak berubah sama sekali. Bagian \textit{core} mana
  yang membuat hal itu mungkin?}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 2: \textit{Error Isolation}}}
  \vspace{3pt}
\begin{lstlisting}[language=bash]
python uji_isolasi.py
# Dengan error handler : 3 perintah jalan, 1 dilewati
# Tanpa error handler  : 3 modul terimpor, lalu berhenti
\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}\small
    \item Tunjukkan baris pada \texttt{plugins/plugin\_rusak.py} yang
          menyebabkan kegagalannya.
    \item Jalankan \texttt{python inti.py rusak} lalu \texttt{python inti.py
          kurs}, dan catat apakah perintah kedua tetap dilayani.
    \item Hapus sementara \textit{error handler} di \texttt{inti.py}, jalankan
          ulang \texttt{python inti.py rusak}, lalu catat akibatnya.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: mengapa kegagalan saat impor lebih berbahaya daripada
  kegagalan saat pemanggilan?}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 3: \textit{Activation Cost}}}
  \vspace{3pt}
\begin{lstlisting}[language=bash]
python ukur_aktivasi.py
# Putaran lazy lebih cepat     : 100 dari 100
\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}\small
    \item Catat kedelapan angka pada laporan akhirnya.
    \item Tunjukkan baris yang membersihkan \textit{import cache}.
    \item Hapus sementara pembersihan itu, jalankan ulang, lalu catat perubahan
          angkanya.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: mengapa selisih pada bab ini layak disebut ada, sedangkan
  selisih pada Bab 2 tidak?}
\end{frame}
