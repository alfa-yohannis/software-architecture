\author{Alfa Yohannis}

\frame{\titlepage}


\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{10pt}
  \begin{enumerate}
    \item Menjelaskan peran processing unit, virtualized middleware, dan
          data pump beserta jalan satu \textit{request} dari masuk sampai
          tersimpan.
    \item Membedakan jalur \textit{request} versi basis data dari versi
          \textit{space}, lalu menentukan kapan masing-masing layak dipilih.
    \item Mengukur \textit{throughput scaling}, \textit{replication lag}
          beserta \textit{data collision}, dan \textit{data loss} pada kedua
          versi.
  \end{enumerate}

  \vspace{10pt}
  {\small Contohnya pengisian rencana studi, yaitu beban yang datang serentak
  dan memperebutkan kuota yang jumlahnya terbatas.}
\end{frame}

\begin{frame}{{\Large Skenario Bisnis}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Pengisian rencana studi dibuka pada jam yang sama bagi seluruh
          angkatan. Ribuan mahasiswa menekan simpan pada menit yang sama, lalu
          sepi lagi sesudahnya.
    \item Satu \textit{request} berisi satu mahasiswa yang mengambil satu
          kelas, dan setiap kelas punya kuota kuota yang terbatas.
    \item Yang diminta pimpinan, kuota tidak boleh terlampaui, jawabannya
          cepat, dan tidak ada pengambilan yang hilang.
    \item Beban seperti inilah yang dijawab \textit{space-based architecture},
          yaitu \textit{concurrency} tinggi yang naik turun tajam.
  \end{itemize}

  \vspace{10pt}
  {\small Richards dan Ford memakai penjualan tiket konser sebagai contoh
  khasnya, dan bentuk bebannya sama persis dengan pengisian rencana studi.}
\end{frame}

\begin{frame}{{\Large Pendahuluan}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Biasanya, jika \textit{request} naik, maka yang ditambahkan adalah
          jumlah \textit{processing unit}-nya. Hal itu dapat dilakukan dengan
          mudah, karena masing-masing unit dapat memproses secara independen.
    \item Yang menjadi \textit{bottleneck} adalah basis datanya. Jika seluruh
          unit menulis ke basis data yang sama, maka terjadi saling menunggu
          giliran menulis, atau \textit{write contention}.
    \item Pada pengukuran bab ini, menambah unit dari satu menjadi delapan
          justru menurunkan hasil dari 1.539,3 menjadi 895,8
          \textit{request} per detik.
    \item Jalan keluarnya, kuota kelas mata kuliah yang dibutuhkan setiap \textit{request}
          di-\textit{copy} ke memori setiap unit, sedangkan penulisan ke basis
          data dikerjakan belakangan.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Istilah yang Dipakai}}
  \vspace{10pt}
  \begin{itemize}\small
    \item \textbf{\textit{Request}} pada konteks ini adalah satu mahasiswa yang
          mengambil satu kelas mata kuliah, misalnya 202600001 mengambil IF231303-A.
    \item \textbf{Kuota} adalah jumlah maksimum mahasiswa di dalam satu
          kelas mata kuliah. Sisanya berkurang satu setiap kali ada yang
          mengambil, dan tidak boleh menjadi minus.
    \item \textbf{Tupel} adalah baris data yang diberi nama di kolom
          pertamanya, yaitu \texttt{("kelas", "IF231303-A", 40, 1)}. Kolomnya
          berturut-turut nama baris, kode kelas mata kuliah, sisa kuotanya, dan
          nomor versi.
    \item \textbf{\textit{Space}} adalah tempat bersama untuk menitipkan dan
          mengambil tupel, dan namanya \textit{tuple space}.
    \item \textbf{Grid} adalah \textit{copy} baris kelas mata kuliah di memori satu unit,
          dan isinya disamakan antar unit lewat replikasi.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{{\Large Jejak Satu \textit{Request}}}
  \vspace{4pt}
\begin{lstlisting}[language=bash]
1. Mahasiswa 202600001 menekan simpan untuk IF231303-A.
2. Messaging grid menerima request, antreannya menjadi 1.
3. unit-1 yang menganggur mengambilnya dari antrean.

9. Data writer menulis 1 kiriman, basis data menjadi 39.
\end{lstlisting}

\begin{frame}{{\Large Membaca Hasilnya}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Mengeluarkan basis data dari jalur \textit{request} membayar paling
          besar, yaitu 1.491,2 menjadi 15.980,5 \textit{request} per detik
          pada satu unit yang sama.
    \item Menambah unit pada versi basis data justru menurunkannya, yaitu
          1.491,2 menjadi 1.049,1. Enam koneksi berebut satu kunci tulis, dan
          yang kalah menerima SQLITE\_BUSY lalu mencoba lagi.
    \item Pada versi \textit{space} penulisnya hanya satu, yaitu \textit{data
          writer} dengan satu koneksi, dan \textit{commit}-nya lima puluh
          perubahan sekaligus. Tidak ada yang berebut.
    \item Menambah unit pada versi \textit{space} juga menurunkannya, 15.980,5
          menjadi 7.651,0, padahal tiap unit sudah memegang prosesor sendiri.
          Bingkai berikutnya memisahkan sebabnya.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Hasil: Beban Banyak Membaca}}
  \vspace{4pt}

\begin{frame}{{\Large Membaca Latensinya}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Latensi dihitung per \textit{request}, yaitu sejak masuk antrean
          sampai mahasiswanya dijawab, sehingga lama menunggu giliran ikut

    \item Pada beban penuh menulis, median versi basis data 471,30 ms

    \item Perebutan terbaca pada nilai terbesar, yaitu 1.018,68 ms pada satu

          menyembunyikannya.
    \item Ragam antar putaran perlu dibaca lebih dahulu. Versi basis data pada
          delapan unit berkisar 1.006,3 sampai 1.348,0 ms, sehingga selisih
          yang lebih kecil daripada itu belum dapat disebut ada.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Rumus Taksiran Tabrakan}}
  \vspace{6pt}
  {\small Richards dan Ford menaksir banyaknya tabrakan per detik dengan
  N $\times$ (UR$^2$ / S) $\times$ RL.}

  \vspace{6pt}
  \begin{itemize}\small
    \item N, \textit{number of instances}, yaitu jumlah unit yang memegang
          copy kuota yang sama.
    \item UR, \textit{update rate}, yaitu banyaknya perubahan kuota per detik
          oleh seluruh unit.
    \item S, \textit{cache size}, yaitu jumlah baris kelas mata kuliah di
          dalam grid.
    \item RL, \textit{replication lag}, yaitu lama satu perubahan sampai
          terpakai unit lain.
  \end{itemize}

  \vspace{6pt}
  {\small Arahnya terbaca dari letaknya. UR berada di pembilang dan
  dikuadratkan, sehingga laju perubahan yang naik dua kali menaikkan taksiran
  empat kali. S berada di penyebut, sehingga baris yang sedikit justru
  menaikkan tabrakan.}
\end{frame}

  \vspace{6pt}
  {\small Tiap keadaan mengubah satu peubah saja, lalu tabrakan nyatanya
  dicacah dan dibandingkan dengan taksiran rumus itu.}

    \hline
    \textbf{Keadaan} & \textbf{N} & \textbf{RL} & \textbf{S} & \textbf{UR} & \textbf{Nyata lawan taksiran} \\
    \hline
    acuan & 5 & 100 ms & 50 & 198 & 203,5 lawan 390,1 per detik \\
    \hline
    lag kecil & 5 & 10 ms & 50 & 198 & 35,0 lawan 39,0 \\
    \hline
    unit sedikit & 2 & 100 ms & 50 & 194 & 57,0 lawan 150,5 \\
    \hline
    baris sedikit & 5 & 100 ms & 10 & 198 & 491,5 lawan 1.950,3 \\
    \hline
    laju dua kali & 5 & 100 ms & 50 & 391 & 664,5 lawan 1.528,8 \\
    \hline
  \end{tabularx}
\end{frame}

\begin{frame}{{\Large Membaca Tabrakannya}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Tabrakan terjadi ketika dua unit mengubah baris kelas mata kuliah yang sama di
          dalam lag, lalu satu perubahan menimpa perubahan yang lain.
    \item Arah rumusnya terbukti, yaitu tabrakan turun bersama lag, naik ketika
          barisnya sedikit, dan naik lebih cepat daripada laju perubahannya.
    \item Laju dinaikkan dua kali, dari 198 menjadi 391 per detik, sedangkan
          tabrakannya naik 3,3 kali. Itulah suku kuadrat pada rumusnya.
    \item Besarannya sendiri sekitar seperempat sampai separuh taksiran rumus,
          sehingga rumus itu layak dipakai sebagai batas atas, bukan sebagai
          ramalan.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Kasus: Kuota Terjual Dua Kali}}
  \vspace{8pt}
  \begin{enumerate}\small
    \item Kelas mata kuliah IF231303-A menyisakan satu kuota menurut grid kedua
          unit.
    \item Dua mahasiswa menekan simpan, lalu kedua \textit{request} masuk ke
          antrean berurutan.
    \item unit-1 mengambil yang pertama, unit-2 mengambil yang kedua. Antrean
          menjamin satu \textit{request} hanya diambil satu unit, namun tidak
          menahan unit yang sudah memegangnya.
    \item Selangnya mikrodetik, sedangkan lag-nya 100 ms. Kedua unit karena itu
          masih membaca sisa satu, lalu keduanya menjawab diterima.
    \item Sesudah lag lewat, perubahan yang satu menimpa yang lain, sehingga
          sisanya tercatat nol, bukan minus satu.
  \end{enumerate}

  \vspace{6pt}
  {\small Akibatnya kelas itu menerima satu mahasiswa melebihi daya
  tampungnya.}
\end{frame}

\begin{frame}{{\Large Kapan Dipakai, Kapan Tidak}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Dipakai ketika \textit{request} datang serentak dan naik turun tajam,
          misalnya pengisian rencana studi atau penjualan tiket.
    \item Tidak dipakai ketika datanya besar dan selalu berubah, sebab
          mereplikasi seluruhnya ke tiap unit justru menjadi beban.
    \item Tidak dipakai pula ketika satu kesalahan hitung tidak boleh terjadi
          sama sekali, sebab tabrakan dan \textit{stale read} memang mungkin.
    \item Dibayar dengan kerumitan, yaitu grid, replikasi, pump, writer,
          reader, dan deployment manager yang semuanya harus dipantau.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large \textit{Summary}: Yang Diubah}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Yang menjadi batas bukan menghitung sisa kuotanya, melainkan
          menuliskannya. Pada versi basis data seluruh unit menulis ke satu
          berkas, sehingga tiap \textit{request} menunggu giliran menulis.
    \item Versi \textit{space} memberi setiap unit satu copy sisa kuota di
          memorinya sendiri, lalu mahasiswa dijawab dari copy itu.
    \item Perubahan yang sudah dijawab dititipkan ke \textit{data pump}, lalu
          \textit{data writer} menuliskannya ke basis data sekelompok
          sekaligus.
    \item Basis data karena itu tidak lagi berada di jalur \textit{request}.
          Mahasiswa menerima jawaban sebelum barisnya tersimpan.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large \textit{Summary}: Yang Dibayar}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Beban 1.200 \textit{request} selesai dijawab dalam 903,3 ms pada
          versi basis data dan 101,4 ms pada versi \textit{space}, yaitu 8,9
          kali lebih cepat pada jumlah unit yang sama.
    \item \textit{Stale read}. Selama \textit{replication lag} belum lewat,
          unit lain masih menjawab dengan sisa kuota yang lama, dan pada lag
          100 ms selisihnya mencapai 48 kuota.
    \item \textit{Data collision}. Dua unit mengubah baris yang sama di dalam
          lag, lalu satu perubahan menimpa yang lain, sehingga satu kuota
          terjual dua kali.
    \item \textit{Data loss}. Perubahan yang masih menunggu di penyangga unit
          lenyap bersama unit yang mati, dan putaran terburuk kehilangan 45
          \textit{request}.
  \end{itemize}

  \vspace{4pt}
  {\small Ketiganya satu pesan, yaitu memindahkan penyimpanan ke luar jalur
  \textit{request} membeli kecepatan dengan membayar ketepatan pada saat yang
  paling buruk.}
\end{frame}
