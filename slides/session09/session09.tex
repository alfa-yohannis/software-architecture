\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{10pt}
  \begin{enumerate}
    \item Menjelaskan pembagian antara \textit{publisher}, event, dan \textit{subscriber} beserta
          kontrak yang menghubungkan ketiganya.
    \item Membedakan topologi broker dari topologi mediator, lalu menentukan
          kapan masing-masing layak dipilih.
    \item Mengukur \textit{latency cost}, \textit{failure containment}, dan
          \textit{change cost} pada ketiga susunan.
  \end{enumerate}

  \vspace{10pt}
  {\small Contohnya satu \textit{request} konversi kurs, dibuat tiga susunan yang
  memanggil \textit{event processor} yang sama persis.}
\end{frame}

\begin{frame}{{\Large Konteks Bisnis}}
  \vspace{8pt}
  \begin{itemize}\small
    \item \textit{Foreign exchange desk} sebuah bank menerima \textit{request}
          konversi sepanjang hari bursa.
    \item Setiap \textit{request} wajib tercatat untuk audit, nilai rupiahnya
          wajib diperiksa, lalu pemohonnya wajib diberitahu.
    \item Kursnya tetap pada contoh bab ini, yaitu USD/IDR 16.250, EUR/IDR
          17.600, dan SGD/IDR 12.100.
    \item \textit{Request} yang nilainya menyamai atau melampaui
          Rp500.000.000 ditandai untuk ditinjau petugas.
  \end{itemize}

  \vspace{8pt}
  {\small Ketiga pekerjaan itu wajib dan urutannya tidak boleh tertukar, sebab
  teks pemberitahuan bergantung pada keputusan \textit{validator}. Batas
  Rp500.000.000 adalah aturan contoh bab ini, bukan kutipan aturan lembaga mana
  pun.}
\end{frame}

\begin{frame}{{\Large Pendahuluan}}
  \vspace{8pt}
  \begin{itemize}\small
    \item Satu \textit{request} konversi kurs perlu dicatat ke riwayat, diperiksa
          nilainya, lalu diberitahukan kepada pemohon.
    \item Cara paling lurus memanggil ketiga pekerjaan itu berurutan dari satu
          tempat, sehingga pemanggilnya terikat pada ketiganya sekaligus.
    \item \textit{Event-driven architecture} menerapkan \textit{decoupling}, yaitu
          pemanggil cukup mengumumkan bahwa sesuatu telah terjadi lalu berhenti.
    \item Pihak yang berkepentingan mendaftarkan diri sendiri, dan pengumuman
          itu sampai kepada mereka tanpa disebut namanya.
    \item Tiga susunan dibandingkan, yaitu susunan sinkron tanpa event sama
          sekali, topologi broker, dan topologi mediator.
  \end{itemize}

  \vspace{8pt}
  {\small Kedua topologi berbasis event pada bab ini mengikuti kerangka Richards
  dan Ford, sedangkan nama pola \textit{message}-nya mengikuti katalog Hohpe dan Woolf.}
\end{frame}

\begin{frame}{{\Large Tiga yang Diukur}}
  \vspace{10pt}
  \begin{itemize}\small
    \item \textbf{\textit{Latency cost}} adalah tambahan waktu satu
          \textit{request} dari masuk sampai seluruh langkahnya tuntas.
    \item \textbf{\textit{Failure containment}} adalah berapa \textit{request}
          tetap tuntas ketika satu \textit{event processor} gagal, dan seberapa
          jauh kegagalannya merembet.
    \item \textbf{\textit{Change cost}} adalah berapa \textit{file} dan baris
          berubah ketika satu \textit{subscriber} baru disisipkan.
  \end{itemize}

  \vspace{10pt}
  {\small Ketiganya diukur pada basis kode yang sama, sehingga selisihnya hanya
  berasal dari cara \textit{message}-nya disampaikan.}
\end{frame}

\begin{frame}{{\Large Istilah yang Dipakai}}
  \vspace{10pt}
  \begin{itemize}\small
    \item \textbf{\textit{Publisher}} mengumumkan bahwa sesuatu telah terjadi, lalu
          selesai tanpa menunggu siapa pun.
    \item \textbf{Event} adalah \textit{message} yang memberitakan kejadian itu
          beserta datanya, atau \textit{event message}.
    \item \textbf{\textit{Subscriber}} menyatakan minat pada satu jenis \textit{event
          message}, lalu mengerjakan bagiannya setiap kali kiriman itu datang.
    \item \textbf{\textit{Topic}} adalah nama saluran tempat \textit{event message} diumumkan, dan setiap
          \textit{subscriber} \textit{topic} tersebut punya antrean sendiri.
  \end{itemize}

  \vspace{10pt}
  {\small Saluran seperti itu disebut \textit{publish-subscribe channel}, sebab
  satu salinan dikirim kepada setiap \textit{subscriber}.}
\end{frame}

\begin{frame}{{\Large Event, Bukan Perintah}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Nama \texttt{request.recorded} menyebut sesuatu yang sudah terjadi,
          sehingga \textit{publisher}-nya tidak menuntut apa pun.
    \item Nama \texttt{record\_request} menyebut sesuatu yang harus
          dikerjakan, sehingga \textit{publisher}-nya menuntut satu pekerjaan tertentu.
    \item Nama kedua memaksa \textit{publisher} tahu bahwa ada pihak yang mencatat, dan
          ikatan itulah yang sebenarnya hendak dilepas.
    \item Penamaan karena itu bukan soal selera, melainkan penentu siapa
          mengenal siapa.
  \end{itemize}

  \vspace{10pt}
  {\small Seluruh \textit{topic} pada bab ini memakai bentuk pertama, yaitu
  \texttt{request.received}, \texttt{request.recorded}, dan
  \texttt{request.validated}.}
\end{frame}

\begin{frame}[fragile]{{\Large Kode: Inti Broker}}

\begin{lstlisting}[style=PythonStyle]
def terbitkan(self, topic: str, pesan: domain.Request) -> int:
  """Menaruh message pada antrean setiap subscriber topic tersebut."""
  antrean_subscriber = self.langganan_topic.get(topic, [])
  for antrean in antrean_subscriber:
    antrean.put(pesan)
  with self.kunci:
    self.jumlah_pesan += len(antrean_subscriber)
  return len(antrean_subscriber)
\end{lstlisting}

  \vspace{4pt}
  {\small Brokernya tidak pernah tahu apa yang dikerjakan penangannya, sehingga
  \textit{topic} tanpa \textit{subscriber} dijawab nol tanpa \textit{error}.}

    \item Mediator memanggil \textit{event processor} pertama, menunggu balasannya, lalu

\begin{frame}[fragile]{{\Large Kode: Urutan Langkah Mediator}}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle]
def jalankan_urutan(self, request: domain.Request) -> bool:
  """Memanggil setiap pemroses bergantian, lalu berhenti pada error pertama."""
  for nomor_langkah, nama in enumerate(self.urutan):
    self.antrean_perintah[nama].put(request)
    self.hitung_pesan(1)
    _, _, galat = self.antrean_balasan.get()
    if galat is not None:
      sisa = self.urutan[nomor_langkah + 1:]
      self.gagal.append((request.nomor, nama, list(sisa)))
      return False
  return True
\end{lstlisting}

  \vspace{2pt}
  {\small Isi mediatornya hanya satu perulangan. Ketika satu langkah gagal, yang
  dicatat bukan hanya nama langkahnya, melainkan juga sisa langkah yang belum
  dijalankan.}
\end{frame}

\begin{frame}{{\Large Analogi: Ketua Panitia Wisuda}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Ketua memegang susunan acara, menelepon seksi konsumsi, menunggu
          laporan selesai, baru menelepon seksi dekorasi.
    \item Bila seksi konsumsi melapor gagal, ketua tahu acara berhenti di
          langkah itu dan tahu seksi mana yang belum dihubungi.
    \item Harganya, setiap kabar melewati satu orang yang sama, dan orang itu
          hanya dapat menangani satu urusan pada satu waktu.
  \end{itemize}

  \vspace{10pt}
  {\small Kedua akibat itu muncul kembali sebagai angka, yaitu catatan langkah
  yang putus pada seksi ketahanan dan waktu tunggu pada seksi berikutnya.}

\end{frame}

\begin{frame}{{\Large Membaca \textit{Latency Cost}}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Angka itu menolak harapan yang sering dibawa ke bab ini, yaitu bahwa
          event membuat \textit{request} lebih cepat.
    \item Median topologi broker 6,2 kali median susunan sinkron, dan topologi
          mediator 9,3 kali.
    \item \textit{Range} jalur sinkron 0,0057 ms, sedangkan selisih median
          terhadap broker 0,0280 ms, yaitu 4,9 kali \textit{range} itu.
    \item Susunan sinkron lebih cepat pada 200 dari 200 putaran, sehingga
          selisihnya bukan kebetulan.
  \end{itemize}

  \vspace{10pt}
  {\small Sebabnya perpindahan antar \textit{thread}. Pekerjaan pada bab ini
  hanya satu perkalian dan dua penyusunan teks, sehingga \textit{cost}
  pengangkutannya justru mendominasi.}
\end{frame}

\begin{frame}{{\Large Kejanggalan \textit{Publisher Blocking}}}
  \vspace{8pt}
  {\small Pada tabel sebelumnya, kolom \textit{blocking} justru dimenangkan
  kedua susunan berbasis event.}

  \vspace{8pt}
  \begin{itemize}\small
    \item \textit{Blocking} broker 0,0024 ms dan sinkron 0,0049 ms, sehingga
          selisihnya 0,0025 ms.
    \item \textit{Range} jalur sinkron 2,3 kali selisih itu, sehingga menurut
          ukuran Bab 2 selisihnya belum dapat disebut ada.
    \item \textit{Win count} lebih peka daripada median, dan arahnya jelas,
          yaitu broker menang pada 199 dari 200 putaran dan mediator pada 200.
  \end{itemize}

  \vspace{8pt}
  {\small Selisih sekecil itu belum menarik. Yang menarik apa yang terjadi
  ketika satu \textit{event processor} mulai lambat.}
\end{frame}

\begin{frame}{{\Large Kasus: Loket Pembayaran}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Loket memakai susunan sinkron, yaitu setiap pembayaran dicatat,
          diperiksa, lalu diberitahukan lewat layanan \textit{message} singkat pihak
          ketiga.
    \item Pada hari biasa layanan itu menjawab dalam puluhan milidetik.
    \item Pada hari terakhir pembayaran, layanan tersebut tersendat dan
          jawabannya berubah menjadi lima detik.
    \item Petugas tidak dapat menerima pembayaran berikutnya sebelum jawaban itu
          datang, antrean mahasiswa mengular, dan sebagian pulang tanpa
          membayar.
  \end{itemize}

  \vspace{10pt}
  {\small Pencatatan dan pemeriksaannya sendiri sebenarnya sudah selesai sejak
  awal, sehingga yang menahan loket adalah langkah terakhirnya.}
\end{frame}

\begin{frame}{{\Large Hasil: \textit{Failure Containment}}}
  \vspace{6pt}
  {\small \textit{Failure containment} adalah berapa \textit{request} tetap
  tuntas ketika satu \textit{event processor} gagal, dan seberapa jauh
  kegagalannya merembet.}

  \vspace{4pt}
  {\small Seratus \textit{request}, \textit{validator}-nya menolak setiap \textit{request} kelipatan
  lima.}

  \vspace{6pt}
  \centering
  \small
  \begin{tabularx}{0.96\textwidth}{|l|l|l|l|l|X|}
    \hline
    \textbf{Susunan} & \textbf{Tuntas} & \textbf{Tercatat} & \textbf{Diberitahu} & \textbf{\textit{Error}} & \textbf{Berhenti} \\
    \hline
    Sinkron & 4 & 5 & 4 & 0 & pada nomor 5 \\
    \hline
    Broker & 80 & 100 & 80 & 20 & tidak \\
    \hline
    Mediator & 80 & 100 & 80 & 20 & tidak \\
    \hline
  \end{tabularx}

  \vspace{6pt}
  \raggedright
  {\small Susunan sinkron berhenti pada \textit{request} kelima, sehingga sembilan
  puluh lima \textit{request} sesudahnya tidak pernah tersentuh. Kedua susunan
  berbasis event menampung \textit{error}-nya di dalam \textit{thread} \textit{subscriber}.}
\end{frame}

\begin{frame}{{\Large Catatan \textit{Error} Kedua Topologi}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Angka keduanya sama, yaitu 80 tuntas dan 100 tercatat, namun catatan
          \textit{error}-nya berbeda.
    \item Broker mencatat \textit{topic} beserta nomor \textit{request}-nya, yaitu
          \texttt{topic request.recorded, request 5}.
    \item Mediator mencatat langkah yang gagal sekaligus sisanya, yaitu
          \texttt{request 5 putus pada validator, sisa ['notifier']}.
    \item Hanya mediator yang memegang urutan lengkapnya, sehingga hanya
          mediator yang dapat mengulang sisa langkah itu kelak.
  \end{itemize}

  \vspace{10pt}
  {\small Itulah yang dibeli dengan empat \textit{message} tambahan per \textit{request}.}
\end{frame}

\begin{frame}{{\Large Hasil: \textit{Publisher Blocking}}}
  \vspace{6pt}
  {\small \textit{Publisher blocking} adalah lama pemanggil terkunci menunggu
  sebelum dapat mengirim \textit{request} berikutnya. Kolom \textit{publisher}
  kembali memuat angkanya.}

  \vspace{4pt}
  {\small Seratus \textit{request}, \textit{validator}-nya menahan lima milidetik. Seluruh
  waktunya untuk seratus \textit{request} sekaligus, bukan per \textit{request}.}

  \vspace{6pt}
  \centering
  \footnotesize
  \begin{tabularx}{0.96\textwidth}{|l|l|l|l|>{\raggedright\arraybackslash}X|}
    \hline
    \textbf{Susunan} & \textbf{\textit{Publisher blocking}} & \textbf{\textit{Peak backlog}} & \textbf{Seluruh tercatat} & \textbf{Seluruh tuntas} \\
    \hline
    Sinkron & 513,1 ms & 0 & 513,4 ms & 513,5 ms \\
    \hline
    Broker & 0,1 ms & 100 & 1,3 ms & 512,0 ms \\
    \hline
    Mediator & 0,1 ms & 100 & 543,9 ms & 548,6 ms \\
    \hline
  \end{tabularx}

  \vspace{6pt}
  \raggedright
  {\small \textit{Peak backlog} adalah jumlah \textit{message} terbanyak yang pernah menumpuk
  pada satu antrean selama percobaan berjalan.}
\end{frame}

\begin{frame}{{\Large Membaca \textit{Publisher Blocking}}}
  \vspace{10pt}
  \begin{itemize}\small
    \item \textit{Blocking} pada \textit{publisher} sinkron 513,1 ms, sedangkan \textit{publisher} pada kedua
          susunan berbasis event kembali sesudah 0,1 ms. Perbandingannya lebih
          dari seribu kali.
    \item Pekerjaan yang tidak lagi ditunggu \textit{publisher} tidak lenyap, melainkan
          menumpuk, dan puncaknya seratus \textit{message}.
    \item Seratus \textit{request} selesai dicatat sesudah 1,3 ms pada topologi
          broker, sedangkan topologi mediator membutuhkan 543,9 ms.
    \item Sebabnya, \textit{recorder} broker punya antrean sendiri dan tidak menunggu
          \textit{validator}, sedangkan mediator menangani satu \textit{request} pada satu
          waktu.
  \end{itemize}

  \vspace{10pt}
  {\small Pengawasan terpusat karena itu bukan hanya menambah \textit{message}, melainkan
  juga membuat mediatornya menjadi leher botol.}
\end{frame}

\begin{frame}{{\Large Hasil: \textit{Change Cost}}}
  \vspace{6pt}
  {\small Bagian audit meminta setiap \textit{request} disalin ke jurnal tersendiri,
  sehingga satu \textit{event processor} baru disisipkan pada ketiga susunan.}

  \vspace{6pt}
  \centering
  \small
  \begin{tabularx}{0.96\textwidth}{|l|l|l|l|X|}
    \hline
    \textbf{Susunan} & \textbf{Baris tambah} & \textbf{Baris hapus} & \textbf{\textit{Message} sebelum} & \textbf{\textit{Message} sesudah} \\
    \hline
    Sinkron & 6 & 3 & 0 & 0 \\
    \hline
    Broker & 3 & 0 & 3 & 4 \\
    \hline
    Mediator & 3 & 1 & 7 & 9 \\
    \hline
  \end{tabularx}

  \vspace{6pt}
  \raggedright
  {\small Selisih barisnya kecil, dan kekecilan itu jujur. Yang lebih
  menentukan letak barisnya, yaitu badan \texttt{kirim} pada susunan sinkron
  dan bagian perakitan pada kedua topologi berbasis event.}
\end{frame}

\begin{frame}{{\Large Urutan Antar \textit{Subscriber}}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Pada topologi mediator, jejak langkahnya selalu \texttt{recorder,
          validator, notifier, audit}.
    \item Pada topologi broker, dua belas kali penjalanan menghasilkan tiga
          bentuk, yaitu \texttt{audit} di urutan pertama delapan kali, kedua
          dua kali, dan terakhir dua kali.
    \item Sebabnya, \textit{recorder} audit berlangganan pada \textit{topic} yang sama dengan
          \textit{recorder} biasa, sehingga keduanya berjalan sejajar di \textit{thread}
          masing-masing.
    \item Akibatnya, urutan antar \textit{subscriber} satu \textit{topic} tidak boleh diandalkan.
  \end{itemize}

  \vspace{10pt}
  {\small \textit{Subscriber} baru juga menambah satu \textit{message} per \textit{request} pada broker dan
  dua \textit{message} pada mediator.}
\end{frame}

\begin{frame}{{\Large Kapan Broker, Kapan Mediator}}
  \vspace{8pt}
  {\small Broker cocok ketika langkah-langkahnya berdiri sendiri dan pemanggil
  tidak boleh menunggu.}
  \begin{itemize}\small
    \item Pesanan toko daring memicu penagihan, pengurangan stok, dan surel
          pemberitahuan, yang tidak saling membutuhkan hasil.
    \item Pendengar baru sering ditambah, misalnya audit dan analitik.
  \end{itemize}

  \vspace{6pt}
  {\small Mediator cocok ketika urutannya mengikat dan harus dapat
  dipertanggungjawabkan.}
  \begin{itemize}\small
    \item Pengajuan kredit wajib melewati penilaian skor, pemeriksaan
          kepatuhan, baru pencairan.
    \item Langkah yang gagal harus diulang atau dibatalkan, sehingga sisa
          langkahnya perlu diketahui.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large \textit{Summary}}}
  \vspace{10pt}
  \begin{itemize}\small
    \item \textit{Publisher} mengumumkan bahwa sesuatu telah terjadi lalu berhenti, dan
          \textit{subscriber} mendaftarkan dirinya sendiri pada \textit{topic} yang diminatinya.
    \item \textit{Decoupling} itu dibayar dengan waktu, yaitu latensi broker 6,2 kali
          latensi sinkron dan mediator 9,3 kali, sedangkan sinkron menang pada
          200 dari 200 putaran.
    \item Yang dibeli terlihat saat satu \textit{event processor} tersendat, yaitu \textit{publisher}
          sinkron mengalami \textit{blocking} 513,1 ms sedangkan \textit{publisher} berbasis event kembali
          sesudah 0,1 ms.
    \item Broker memakai tiga \textit{message} dan melepas langkah pertama dari langkah
          lambat di belakangnya, sedangkan mediator memakai tujuh \textit{message} namun
          tahu langkah mana yang putus beserta sisanya.
  \end{itemize}

  \vspace{10pt}
  {\small Yang berpindah adalah tempat menunggu. Waktu tunggu \textit{publisher} kini
  menumpuk di antrean, dan tumpukan seratus \textit{message} itulah yang harus
  dipantau.}
\end{frame}
