\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{10pt}
  \begin{enumerate}
    \item Menjelaskan pemisahan sisi \textit{command} dari sisi \textit{query} beserta \textit{projection}
          yang menjembatani keduanya.
    \item Membedakan \textit{storage} \textit{current state} dari \textit{storage} event,
          lalu menentukan kapan masing-masing layak dipilih.
    \item Mengukur \textit{write cost} dan \textit{read cost} ketiga versi,
          \textit{replay cost} terhadap panjang \textit{event log}, dan jumlah
          \textit{temporal query} yang terjawab.
  \end{enumerate}

  \vspace{10pt}
  {\small Contohnya buku besar saldo dompet mahasiswa, ditulis tiga kali dengan
  perilaku luar yang sama persis.}
\end{frame}

\begin{frame}{{\Large Pendahuluan}}
  \vspace{8pt}
  \begin{itemize}\small
    \item Bab sebelumnya memakai satu model data untuk menulis sekaligus
          membaca, yaitu bentuk CRUD.
    \item \textit{Command} menuntut keadaan terakhir beserta aturan bisnisnya, sedangkan
          \textit{query} menuntut bentuk yang siap ditampilkan.
    \item CQRS memisahkan \textit{write model} dari \textit{read model}. \textit{Event sourcing}
          menyimpan event, bukan keadaan.
  \end{itemize}

  \vspace{8pt}
  {\small Domainnya dompet mahasiswa dengan dua \textit{command} saja, yaitu setor dan
  bayar. Saldonya tidak boleh minus, dan aturan itu diperiksa sisi \textit{command}.}
\end{frame}

\begin{frame}[fragile]{{\Large \textit{Command} Menambah \textit{Row}}}

def jalankan(self, command: domain.Command) -> int:
  """Menambah satu row event sesudah aturan bisnisnya dipenuhi."""
  _, ringkasan = self.susun_ringkasan(command.pemilik)
  saldo_baru = domain.periksa_command(ringkasan.saldo, command)
  self.koneksi.execute(
    "INSERT INTO event (pemilik, jenis, nominal, waktu) "
    "VALUES (?, ?, ?, ?)",
    (command.pemilik, command.jenis, command.nominal, command.waktu),
  )
  self.koneksi.commit()
  return saldo_baru
\end{lstlisting}
  \vspace{4pt}
  {\small Tidak ada kolom saldo yang dapat dibaca langsung, sehingga saldo untuk
  memeriksa aturan bisnis disusun ulang lebih dahulu.}

\begin{frame}{{\Large Analogi: Papan Skor dan Buku Besar}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Petugas meja pertandingan basket menulis setiap angka pada lembar
          skor resmi, lengkap dengan pencetak dan menitnya.
    \item Penonton yang ingin tahu kedudukan cukup melihat papan skor, dan
          tidak mengganggu petugas itu.
    \item Papan skor diperbarui sesudah lembar skor ditulis, bukan pada saat
          yang sama persis.
    \item Buku besar koperasi menulis setiap transaksi pada baris baru, dan
          saldo akhir halaman ditulis di bagian bawah halaman.
  \end{itemize}

  \vspace{10pt}
  {\small Angka di bawah halaman itulah snapshotnya. Bila diragukan, seluruh
  barisnya masih ada dan dapat dijumlah ulang.}
\end{frame}

\begin{frame}{{\Large Kasus: \textit{Double Debit}}}
  \vspace{8pt}
  \begin{enumerate}\small
    \item Unit keuangan menjalankan dompet mahasiswa di atas satu tabel
          keadaan.
    \item Pekerjaan pemotongan tagihan berjalan dua kali karena percobaan ulang
          yang gagal terdeteksi.
    \item Pimpinan menanyakan siapa saja yang terpotong dua kali dan berapa
          saldo mereka sebelumnya.
    \item Tabelnya hanya memuat saldo terakhir, sehingga kedua pertanyaan itu
          tidak terjawab dari sistem.
  \end{enumerate}

  \vspace{8pt}
  {\small Bukti transfer diminta dari mahasiswa satu per satu, lalu dicocokkan
  dengan tangan selama dua minggu.}
\end{frame}

    CRUD & 12,0 KiB & tidak terjawab & tidak terjawab \\
    \hline
    CQRS & 20,0 KiB & tidak terjawab & 95 \\
    \hline
    \textit{Event sourcing} & 668,0 KiB & 2.132.000 & 95 \\
    \hline
  \end{tabularx}

  \vspace{6pt}
  \raggedright
  {\small CQRS menjawab 95 karena \texttt{ringkasan\_baca} punya kolom
  pencacah \texttt{jumlah\_perubahan}, yaitu pertanyaannya sudah diduga saat
  merancang. Saldo pukul 08.30 tidak diduga, sehingga tidak terjawab.
  \textit{Event sourcing} menjawab keduanya tanpa menduga, dengan harga 55,7
  kali \textit{file} keadaan.}
\end{frame}

\begin{frame}{{\Large \textit{Summary}}}
  \vspace{10pt}
  \begin{itemize}\small
    \item CQRS memisahkan \textit{write model} dari \textit{read model}, dan
          \textit{projection} yang menjembatani keduanya.
    \item Pemisahan itu berbayar, yaitu satu \textit{command} menuntut dua kali simpan,
          56,2983 ms melawan 28,0638 ms.
    \item \textit{Event sourcing} menyimpan event, sehingga riwayatnya
          utuh, tetapi bacanya naik menjadi 0,1776 ms.
    \item Snapshot menekan penghitungan ulang, yaitu 287,7 kali lebih cepat pada
          seratus ribu event.
  \end{itemize}

  \vspace{10pt}
  {\small Penimpaan satu \textit{row} adalah keputusan yang memilih
  kemurahan tulis dan membuang riwayat.}
\end{frame}
