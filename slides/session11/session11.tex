\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{10pt}
  \begin{enumerate}
    \item Menjelaskan tiga ciri service-based architecture, yaitu layanan
          \textit{coarse-grained}, \textit{shared database}, dan \textit{API gateway}.
    \item Membedakan pembagian \textit{coarse-grained} tersebut dari monolit ber-\textit{layer}
          maupun dari mikroservis, lalu menentukan kapan masing-masing layak
          dipilih.
    \item Mengukur tambahan \textit{response time} akibat \textit{hop} jaringan, \textit{request}
          yang selamat ketika satu layanan mati, dan biaya mengubah skema
          \textit{shared database}.
  \end{enumerate}

  \vspace{10pt}
  {\small Bab ini menempati ruang antara monolit dan mikroservis, yaitu bentuk
  yang paling sering ditemui pada \textit{production system}.}
\end{frame}

\begin{frame}{{\Large Pendahuluan}}
  \vspace{8pt}
  \begin{itemize}\small
    \item Pemecahan aplikasi sering dibicarakan sebagai pilihan antara dua
          \textit{extreme} saja, yaitu monolit dan mikroservis.
    \item Service-based architecture menempati ruang antara keduanya,
          lazimnya dengan empat sampai dua belas layanan \textit{domain}. Contoh
          bab ini memakai tiga.
    \item Setiap layanan berjalan sebagai proses tersendiri dan dipanggil
          lewat HTTP.
    \item Biasanya, basis datanya tetap satu dan dipakai bersama, dan ciri
          itulah yang
          membedakannya dari mikroservis.
  \end{itemize}

  \vspace{8pt}
  {\small \textit{Cost}-nya dapat diukur dari tiga arah, yaitu \textit{response time}, \textit{failure resilience}, dan \textit{change cost}.}
\end{frame}

\begin{frame}[fragile]{{\Large Monolit: Pemanggilan Fungsi}}
  \vspace{20pt}
  {\small Pekerjaan yang diukur adalah menerbitkan \textit{invoice} satu \textit{order}.
  \textit{Invoice}-nya menuntut tiga \textit{domain} sekaligus, yaitu \textit{order} untuk mata
  uangnya, kurs untuk nilainya, lalu \textit{billing} untuk menerbitkannya.}

  \vspace{3pt}
  {\small Pada versi monolit, ketiga \textit{domain} berada di dalam satu proses
  di \textit{port} 8000, sehingga jalur gabungannya tiga baris saja. Versi inilah
  \textit{benchmark} bagi versi service-based.}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle]
@aplikasi.post("/pesanan/<int:nomor>/tagihan")
def terbitkan_tagihan(nomor: int) -> dict:
  pesanan = domain_pemesanan.ambil_pesanan(nomor)
  kurs = domain_kurs.ambil_kurs(pesanan["mata_uang"])
  tagihan = domain_penagihan.terbitkan_tagihan(nomor, kurs["nilai"])
  return {"pesanan": pesanan, "kurs": kurs, "tagihan": tagihan,
          "hop": alamat.HOP_KLIEN}
\end{lstlisting}
  \vspace{2pt}
  {\small Tidak ada \textit{request} yang perlu disusun, tidak ada jawaban yang
  perlu diuraikan, dan \textit{hop} jaringannya satu.}
\end{frame}

\begin{frame}[fragile]{{\Large Service-Based: Tiga Panggilan}}
  \vspace{4pt}
  {\small Jalur yang sama dengan versi monolit, namun ketiga pemanggilan
  fungsinya berubah menjadi tiga \textit{request} HTTP ke proses lain.}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle, emph={minta},
  emphstyle={\color{praditagreen}\bfseries}]
@aplikasi.post("/pesanan/<int:nomor>/tagihan")
def terbitkan_tagihan(nomor: int) -> dict:
  pesanan = minta(alamat.PORTA_PEMESANAN, f"/pesanan/{nomor}")
  kurs = minta(alamat.PORTA_KURS, f"/kurs/{pesanan['mata_uang']}")
  tagihan = minta(alamat.PORTA_PENAGIHAN, f"/tagihan/{nomor}", "POST",
                  {"nilai_kurs": kurs["nilai"]})
  return {"pesanan": pesanan, "kurs": kurs, "tagihan": tagihan,
          "hop": (alamat.HOP_KLIEN + pesanan["hop"]
                       + kurs["hop"] + tagihan["hop"])}
\end{lstlisting}
  \vspace{2pt}
  {\small Ketiganya berurutan, sebab nilai kurs baru diketahui sesudah
  \textit{order}-nya terbaca. Angka \textit{hop} dijumlahkan dari jawaban setiap layanan,
  bukan ditulis sebagai tetapan.}
\end{frame}

\begin{frame}{{\Large Penjelasan \textit{Sequence Diagram}}}
  \vspace{8pt}
  \begin{itemize}\small
    \item Satu \textit{request} menerbitkan \textit{invoice}, dan pekerjaan
          \textit{domain}-nya sama persis pada kedua versi.
    \item Versi monolit menyelesaikannya dengan satu \textit{hop}, yaitu dari
          klien ke proses itu saja.
    \item Versi service-based menempuh empat \textit{hop}, sebab
          \textit{gateway} memanggil tiga layanan berurutan.
    \item Ketiganya tidak dapat disejajarkan, sebab kursnya baru diketahui
          sesudah \textit{order}-nya terbaca.
  \end{itemize}

  \vspace{8pt}
  {\small Selisih tiga \textit{hop} itulah yang muncul sebagai tambahan
  \textit{response time} pada seksi hasil.}
\end{frame}

\begin{frame}[fragile]{{\Large \textit{Cross-Domain Read}}}
  \vspace{4pt}
\begin{lstlisting}[style=PythonStyle, escapeinside={(*@}{@*)}]
AMBIL_PESANAN_UNTUK_TAGIHAN = """
SELECT nomor, nominal, status
(*@\colorbox{praditagreen!30}{\ttfamily\scriptsize FROM pesanan}@*)
WHERE nomor = ?
"""
    if baris["status"] == basis_data.STATUS_BATAL:
      raise kesalahan.AturanDilanggar(
        f"pesanan {nomor_pesanan} sudah dibatalkan, tidak dapat ditagih")
\end{lstlisting}
  \vspace{2pt}
  \begin{itemize}\small
    \item Baris yang disorot adalah \textit{cross-domain read}: layanan
          \textit{billing} membaca tabel milik \textit{domain} \textit{order} langsung,
          bukan bertanya kepadanya.
    \item Untungnya, satu \textit{query} sudah cukup, tanpa \textit{hop}
          jaringan dan tanpa data ganda.
    \item Akibatnya, nama kolom pada tabel \texttt{pesanan} kini dipakai dua \textit{domain}
          sekaligus.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Analogi: Tiga Loket, Satu Lemari}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Kantor layanan mahasiswa punya tiga loket, yaitu keuangan,
          akademik, dan kemahasiswaan.
    \item Ketiga loket melayani antrean sendiri-sendiri, sehingga loket yang
          tutup tidak menghentikan loket sebelahnya.
    \item Dokumen mahasiswa tetap disimpan di satu lemari arsip di belakang
          ruangan.
    \item Ketiga petugas mengambil dari lemari yang sama.
  \end{itemize}

  \vspace{10pt}
  {\small Menutup satu loket memotong satu antrean. Mengunci lemari arsip
  menghentikan ketiga loket sekaligus.}
\end{frame}

\begin{frame}{{\Large Kasus: Satu Kolom Diganti Namanya}}
  \vspace{8pt}
  \begin{enumerate}\small
    \item Sebuah tim memecah sistem pembayaran kampus menjadi empat layanan
          dengan satu basis data.
    \item Pada suatu malam, kolom \texttt{nominal} pada tabel \texttt{pesanan} diganti
          menjadi \texttt{nominal\_asing}.
    \item Perubahan itu dijalankan bersama pembaruan layanan \textit{order} saja.
    \item Layanan \textit{billing} masih memuat \textit{query} lama yang menyebut kolom lama.
  \end{enumerate}

  \vspace{8pt}
  {\small Kegagalannya baru ketahuan keesokan pagi, saat \textit{invoice} pertama
  diterbitkan. Nama kolomnya sudah berganti, sehingga \textit{query} layanan
  \textit{billing} mencari kolom yang sudah tidak ada.}

  \vspace{3pt}
  {\small Skemanya tidak bisa dikembalikan begitu saja, sebab layanan \textit{order}
  versi baru sudah menulis dengan nama kolom baru. Jalan paling cepat adalah
  membatalkan pembaruan layanan \textit{order}, padahal kode layanan itu sendiri tidak
  ada yang salah.}
\end{frame}

\begin{frame}{{\Large \textit{Summary}}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Service-based architecture memecah aplikasi menjadi sedikit layanan
          \textit{coarse-grained}, sedangkan basis datanya biasanya tetap satu.
    \item Karena basis datanya belum dipisah, aturan \textit{cross-domain} masih
          dapat ditulis sebagai satu \textit{query}.
    \item Pemanggilan yang tadinya fungsi kini menjadi \textit{hop} jaringan,
          sehingga \textit{response time} naik mengikuti jumlah \textit{hop}-nya.
    \item Kegagalan satu layanan hanya memutus \textit{request} yang
          membutuhkannya, namun \textit{shared database} tetap dapat menjatuhkan
          semuanya.
  \end{itemize}

  \vspace{10pt}
  {\small Perubahan di dalam satu \textit{domain} menjadi murah, sedangkan
  perubahan pada skema bersama tetap menyentuh beberapa layanan sekaligus.}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 1: Tambahan \textit{Hop}}}
  \vspace{2pt}
  {\small Tujuan: mengukur tambahan \textit{response time} ketika tiga
  pemanggilan fungsi berubah menjadi tiga \textit{request} HTTP.}
  \vspace{2pt}
\begin{lstlisting}[language=bash]
python ukur_latensi.py
# Median monolit                : 30.01 ms
# Median service-based          : 34.70 ms
# Putaran monolit lebih cepat   : 140 dari 150
\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}\small
    \item Catat median, persentil ke-95, dan nilai terbesar kedua versi.
    \item Catat kedua angka \textit{hop} beserta \textit{win count}-nya.
    \item Hitung berapa kali fungsi \texttt{minta} dipanggil pada jalur
          terbitkan \textit{invoice}, lalu bandingkan dengan angka \textit{hop}-nya.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: mengapa selisihnya layak disebut ada walaupun
  \textit{range} jalur monolit lebih besar daripada selisih itu sendiri?}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 2: \textit{Failure Resilience}}}
  \vspace{2pt}
  {\small Tujuan: membuktikan bahwa mematikan satu layanan hanya memutus
  \textit{request} yang membutuhkannya, sedangkan mengunci \textit{shared
  database} memutus semuanya.}
  \vspace{2pt}
\begin{lstlisting}[language=bash]
python uji_kegagalan.py
# Domain kurs dimatikan          0 dari 4    2 dari 4
# Shared database dikunci        0 dari 4    0 dari 4
\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}\small
    \item Catat ketiga baris ringkasan beserta nama \textit{request} yang masih
          terlayani.
    \item Tunjukkan baris pada \texttt{uji\_kegagalan.py} yang mematikan satu
          layanan saja.
    \item Ubah \texttt{LAYANAN\_DIMATIKAN} menjadi \texttt{"pemesanan"},
          jalankan ulang, lalu catat perubahan angkanya.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: apa yang dibuktikan baris basis data terkunci tentang
  batas pemecahan proses?}
\end{frame}

\begin{frame}[fragile]{{\Large Latihan 3: \textit{Change Cost}}}
  \vspace{2pt}
  {\small Tujuan: menghitung berapa layanan harus diproses ulang ketika skema
  \textit{shared database} berubah.}
  \vspace{2pt}
\begin{lstlisting}[language=bash]
python biaya_perubahan.py
#   Service-based  file 3   baris 16   proses ulang 3 dari 4
#   Proses ulang   : kurs, pemesanan, penagihan
\end{lstlisting}

  \vspace{3pt}
  \begin{enumerate}\small
    \item Catat keenam angka pada kedua bagian laporannya.
    \item Baca isi \texttt{PERUBAHAN} untuk melihat penggantian teks yang
          dipakainya.
    \item Terapkan perubahan kedua sungguhan, lalu catat apa yang terjadi bila
          hanya satu dari ketiga \textit{file} yang diubah.
  \end{enumerate}

  \vspace{3pt}
  {\small Pertanyaan: dengan menggabungkan hasil ketiga latihan, kapan
  pembagian \textit{coarse-grained} ini lebih tepat daripada monolit maupun
  mikroservis?}
\end{frame}
