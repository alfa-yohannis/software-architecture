\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{10pt}
  \begin{enumerate}
    \item Menjelaskan taksonomi empat \textit{layer} layanan beserta tugas
          \textit{orchestration engine} yang merangkainya.
    \item Membedakan \textit{workflow} yang tertulis pada satu tempat dari
          \textit{workflow} yang tersebar di dalam layanan.
    \item Mengukur tambahan \textit{hop}, \textit{change cost} saat menyisipkan
          satu langkah, dan jumlah \textit{workflow} yang terdampak ketika satu
          layanan berubah.
  \end{enumerate}

  \vspace{10pt}
  {\small Contoh bab ini adalah pemesanan valuta asing enam langkah, dijalankan
  dengan dua cara yang hasilnya sama persis.}
\end{frame}

\begin{frame}{{\Large Pendahuluan}}
  \vspace{8pt}
  \begin{itemize}\small
    \item \textit{Orchestration-driven service-oriented architecture} (SOA)
          menyusun layanan secara \textit{layered}, lalu merangkainya dengan
          satu \textit{orchestration engine}.
    \item \textit{Engine} itu memegang urutan langkah setiap proses bisnis, atau
          \textit{workflow}. Layanannya tidak saling mengenal.
    \item Model acuannya disusun Organization for the Advancement of Structured
          Information Standards (OASIS).
    \item Pemegang urutan itu disebut \textit{process manager}, sebab ia
          menyimpan urutan langkah sekaligus \textit{state} setiap proses.
  \end{itemize}

  \vspace{8pt}
  {\small Bedanya dari Bab 11 dan Bab 13 ada dua, yaitu \textit{maximum reuse}
  (satu layanan dipakai sebanyak mungkin \textit{workflow}) dan seluruh
  \textit{workflow} yang tertulis di dalam \textit{engine}.}
\end{frame}

\begin{frame}{{\Large Penjelasan Empat \textit{Layer}}}
  \vspace{10pt}
  \begin{itemize}\small
    \item \textit{Business service} adalah pintu masuk proses bisnis, berisi
          urutan langkah saja, misalnya kelima \textit{workflow}.
    \item \textit{Orchestration engine} adalah komponen yang bertanggung jawab
          menjalankan urutan itu dan mencatat \textit{state} proses.
    \item \textit{Enterprise service} adalah layanan yang bertanggung jawab atas
          aturan bisnis bersama, misalnya nasabah dan saldo.
    \item \textit{Application service} adalah layanan yang bertanggung jawab atas
          kebutuhan satu aplikasi saja, misalnya bukti.
    \item \textit{Infrastructure service} adalah layanan yang bertanggung jawab
          atas urusan teknis tanpa aturan bisnis, misalnya audit.
  \end{itemize}

  \vspace{6pt}
  {\small Seluruh panah berangkat dari \textit{engine}, tanpa panah antar
  layanan.}
\end{frame}

\begin{frame}{{\Large Penjelasan \textit{Class Diagram}}}
  \vspace{8pt}
  \begin{itemize}\small
    \item \texttt{DEFINISI\_ALUR} adalah \textit{business service} yang
          bertanggung jawab menyimpan kelima \textit{workflow} sebagai daftar
          nama langkah.
    \item \texttt{OrchestrationEngine} adalah kelas yang bertanggung jawab
          membaca daftar itu, memanggil layanan, dan menyimpan \textit{state}
          proses pada \texttt{riwayat}.
    \item Ketujuh kotak lainnya adalah layanan, dikelompokkan menurut
          \textit{layer}-nya. Angka di dalamnya adalah jumlah \textit{workflow}
          pemakainya.
    \item Setiap panah ke layanan adalah \texttt{POST /langkah} lewat
          Hypertext Transfer Protocol (HTTP).
  \end{itemize}

  \vspace{4pt}
  {\small Tidak ada panah antar layanan, sehingga menambah langkah cukup
  menyunting \texttt{DEFINISI\_ALUR}.}

\begin{frame}[fragile]{{\Large Orkestrasi: Kode}}
  \vspace{8pt}
  {\small Inti \textit{engine} hanya satu perulangan atas
  \texttt{DEFINISI\_ALUR}.}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle, emph={DEFINISI_ALUR, panggil_layanan},
  emphstyle={\color{praditagreen}\bfseries}]
urutan = DEFINISI_ALUR[proses["alur"]]
for nama_langkah in urutan[len(proses["selesai"]):]:
  try:
    tambahan = self.panggil_layanan(nama_langkah, proses["konteks"])
    proses["konteks"].update(tambahan)
  except LayananPutus as kesalahan:
    proses["tertahan"] = nama_langkah
    proses["keadaan"] = "tertahan"
    proses["pesan"] = str(kesalahan)
    return proses
\end{lstlisting}
  \vspace{2pt}
  {\small Setiap langkah yang berhasil dicatat ke daftar \texttt{selesai}, lalu
  perulangan berikutnya dimulai dari langkah yang belum selesai. Proses yang
  tertahan karena itu dapat dilanjutkan tanpa mengulang langkah sebelumnya.}
\end{frame}

\begin{frame}{{\Large Analogi: Kepala Dapur}}
  \vspace{10pt}
  {\small Dapur sebuah rumah makan sedang menyiapkan pesanan nasi goreng
  spesial.}

  \vspace{6pt}
  \begin{itemize}\small
    \item Kepala dapur memegang kartu resepnya: goreng nasi, goreng telur mata
          sapi, tambahkan kerupuk dan acar, lalu tata di piring saji.
    \item Juru masak di tiap pos hanya menguasai satu pekerjaan, misalnya pos
          wajan menggoreng nasi dan pos telur menggoreng telur.
    \item Sesudah satu pos selesai, piringnya kembali ke meja kepala dapur, lalu
          dikirim ke pos berikutnya sesuai kartu resep.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Analogi: Kartu Resep}}
  \vspace{10pt}
  \begin{itemize}\small
    \item Perjalanan piring lebih panjang daripada bila piring diserahkan
          langsung ke pos sebelah.
    \item Akan tetapi kepala dapur selalu tahu piring meja nomor berapa sedang
          berada di pos mana.
    \item Bila menu itu kini memakai bawang goreng, langkah baru cukup ditulis
          pada kartu resep tanpa memberi tahu juru masak lain.
  \end{itemize}

  \vspace{8pt}
  {\small Kepala dapur adalah \textit{orchestration engine}, kartu resep adalah
  \texttt{DEFINISI\_ALUR}, pos adalah layanan, dan perjalanan pulang ke meja
  kepala dapur adalah \textit{hop} tambahannya.}
\end{frame}

\begin{frame}{{\Large Kasus: Terpotong Dua Kali}}
  \vspace{8pt}
  \begin{enumerate}\small
    \item Sebuah bank menjalankan pemesanan valuta asing lewat rangkaian
          layanan yang saling memanggil, tanpa \textit{orchestration engine}.
    \item Pada suatu sore layanan \texttt{transaksi} mati karena
          \textit{disk}-nya penuh.
    \item Dua puluh pesanan berhenti tepat sesudah saldo nasabah terpotong.
    \item Layanannya dinyalakan kembali, lalu petugas menjalankan ulang kedua
          puluh pesanan dari awal, sebab tidak ada catatan langkah mana yang
          sudah selesai.
  \end{enumerate}

  \vspace{8pt}
  {\small Saldo kedua puluh nasabah terpotong untuk kedua kalinya, dan
  pengembaliannya baru selesai tiga hari kemudian.}
\end{frame}

\begin{frame}{{\Large \textit{Summary}}}
  \vspace{12pt}
  \begin{itemize}\small
    \item \textit{Orchestration-driven} SOA menyusun layanan menjadi empat
          \textit{layer}, lalu merangkainya dengan satu \textit{orchestration
          engine} yang memegang urutan langkah dan \textit{state} proses.
    \item Urutan yang tertulis di satu tempat membuat langkah baru dapat
          disisipkan tanpa menyunting layanan yang sedang berjalan.
    \item \textit{State} yang dicatat sesudah setiap langkah membuat proses yang
          putus dapat dilanjutkan, tanpa mengulang langkah yang sudah mengubah
          uang.
    \item \textit{Cost}-nya ada dua, yaitu \textit{hop} tambahan ke
          \textit{engine} pada setiap langkah, dan pengujian ulang yang meluas
          setiap kali layanan bersama berubah.
  \end{itemize}

  \vspace{2pt}
  {\small Orkestrasi terpusat cocok ketika proses bisnisnya sering berubah dan
  wajib dapat dilanjutkan. Bentuk ini menjadi mahal ketika satu layanan dipakai
  terlalu banyak \textit{workflow}.}
\end{frame}
